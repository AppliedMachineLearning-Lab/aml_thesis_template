%  titlepage.tex -- the title page.
%
%  Fill in the block of fields directly below. The layout underneath it should
%  not need editing: it is here, in text/, rather than hidden inside
%  aml-thesis.cls, precisely so that you CAN adjust it if your examination
%  office asks for something different.
%
%  Rows you do not need (a second examiner, the birth line, an affiliation)
%  can simply be dropped -- see the notes at each one.
%
%  Everything on this page except the title is set a step below body size, so
%  that the logos can be large without crowding the page. The vertical gaps
%  below are tuned for that, for 12pt Palatino, and for a title of about four
%  lines. If you change the body font or size, or use a much longer title,
%  check that the page still fits: when it overflows it does so silently, by
%  pushing the examiner block and date onto a second page.
% =============================================================================

% ---- Fill these in ----------------------------------------------------------
\newcommand{\thesiskind}{Master's Thesis}   % or: Bachelor's Thesis, Manual, ...
\newcommand{\thesistitle}{Towards Destroying the Ring in the Fires of Mount Doom}

\newcommand{\thesisauthor}{Gandalf Grey}
\newcommand{\authorbirth}{born 1 January 1900 in City, Country}
\newcommand{\authormatric}{0000000}

% The institution and lab are the same for every thesis written here; the
% degree course is not. Change all three if your programme differs.
\newcommand{\thesisinstitution}{Rheinische Friedrich-Wilhelms-Universit\"{a}t Bonn}
\newcommand{\thesislab}{Applied Machine Learning Lab}
\newcommand{\thesisdegree}{Computer Science (M.Sc.)}

% Examiners and supervisors. This is the body of a two-column table: a label,
% then \person{Name}{Affiliation}, with rows separated by \\ as usual.
%
%   * The affiliation is OPTIONAL. Pass {} and the second line is omitted
%     entirely, leaving no gap:  \person{Prof.\ Dr.\ Example}{}
%   * For several supervisors, give one row each and leave the label empty on
%     the continuation rows so the names stack under a single heading.
%   * Add or delete rows freely -- just do not leave a trailing \\ after the
%     last one, or the table grows an empty row.
\newcommand{\thesispeople}{%
  First Examiner:        & \person{Prof.\ Dr.\ Rafet Sifa}{}   \\
  Second Examiner: & \person{Dr.\ Lorenz Sparrenberg}{}   \\
  Supervisors:     & \person{Prof.\ Dr.\ Rafet Sifa}{}         \\
                   & \person{Dr.\ Tobias Deußer}{}%
}

\newcommand{\thesisdate}{\today}            % or a fixed date: {1 September 2000}

% ---- Logos ------------------------------------------------------------------
% Both are set to the same height so they sit on a common baseline; the widths
% follow from each file's aspect ratio. Change \logoheight to scale both at
% once rather than editing the two lines separately.
\newcommand{\logoheight}{20mm}
\newcommand{\logolab}{\includegraphics[height=\logoheight]{figures/template/aml_lab_tight}}
\newcommand{\logouniversity}{\includegraphics[height=\logoheight]{figures/template/uni_bonn_logo}}

% =============================================================================
%  Layout -- adjust only if you need to.
% =============================================================================
\begin{titlepage}
% The class sets \linespread{1.25} for the body text; that is too airy for a
% title page and pushes the examiner block onto a second page. Reset it for
% the duration of this page only -- the titlepage environment is a group, so
% the body is unaffected.
\setstretch{1}
\centering

% Logos, both flush right: university first, lab to its right.
% \makebox pins the line to exactly \linewidth, which neutralises the
% \centering above -- without it the surrounding centring glue competes with
% the \hfill and the pair drifts back towards the middle.
\noindent\makebox[\linewidth]{\hfill\logouniversity\hspace{5mm}\logolab}\par

\null\vfil

% Title block, set to a golden-ratio fraction of the page width so the rules
% stop well short of the margins.
\begin{minipage}{0.618034\paperwidth}
  \centering
  {\large\textsc{\thesiskind}\par}\vspace{1ex}
  \rule{\linewidth}{1pt}\\
  {\Huge\bfseries\thesistitle\par}
  \rule{\linewidth}{1pt}\vspace{0.8cm}
\end{minipage}\par

% Author.
{\small\emph{written by}\\[1em]
 \large
 \begin{tabular}[t]{c}
   \thesisauthor \\
   % Delete the next line if your regulations do not ask for a birth date.
   \small\authorbirth \\
   \small Matriculation number: \authormatric
 \end{tabular}\par}\vspace{1em}

% Where the thesis is submitted, and under which programme.
{\small\emph{submitted to}\\[0.6em]
 \normalsize
 \begin{tabular}[t]{c}
   \thesisinstitution \\
   \thesislab
 \end{tabular}\par}\vspace{1em}

{\small\emph{in degree course}\\[0.6em]
 \normalsize\thesisdegree\par}\vspace{0.8cm}

% Examiners and supervisors, set smaller than the author's name: these are
% attributions, not the subject of the page.
%
% \person{Name}{Affiliation}: with an empty affiliation this is just the name,
% so no \makecell and therefore no phantom second line.
\newcommand{\person}[2]{\ifblank{#2}{#1}{\makecell[tl]{#1\\\footnotesize #2}}}
{\renewcommand{\arraystretch}{1.3}
 \small
 \begin{tabular}{r@{\quad}l}
   \thesispeople
 \end{tabular}\par}

% \vfil here, matching the \vfil above: equal-strength glue top and bottom
% centres the block vertically. A \vfill (stronger) would absorb all the
% stretch and push everything to the top of the page.
\vfil\null
{\small\thesisdate\par}

\end{titlepage}
